  // ---------- installed ground-mounted solar array ----------
  const solarGeo = [];
  const arrayX = -13.0;
  const arrayZ = -3.5;
  const arrayYaw = 0.35;

  // 1.2 m square base plate, four anchor bolts
  const plate = new RoundedBoxGeometry(1.35, 0.16, 1.35, 2, 0.03);
  plate.translate(arrayX, 0.08, arrayZ);
  dark.push(plate);
  for (const bx of [-0.48, 0.48]) {
    for (const bz of [-0.48, 0.48]) {
      const bolt = new THREE.CylinderGeometry(0.055, 0.055, 0.1, 8);
      bolt.translate(arrayX + bx, 0.18, arrayZ + bz);
      titanium.push(bolt);
    }
  }
  // 0.18 m titanium mast + tracker motor at the top
  const mastA = new THREE.CylinderGeometry(0.16, 0.18, 2.7, 14);
  mastA.translate(arrayX, 1.45, arrayZ);
  titanium.push(mastA);
  const motor = new RoundedBoxGeometry(0.36, 0.28, 0.36, 2, 0.02);
  motor.translate(arrayX, 2.82, arrayZ);
  titanium.push(motor);
  const motorCap = new THREE.CylinderGeometry(0.13, 0.13, 0.12, 12);
  motorCap.rotateX(HALF_PI);
  motorCap.translate(arrayX, 2.82, arrayZ + 0.2);

  // junction box and warning placard at the base
  const junction = new RoundedBoxGeometry(0.48, 0.62, 0.32, 2, 0.02);
  junction.translate(arrayX + 0.45, 0.5, arrayZ + 0.62);

  titanium.push(placard);

  // 2 rows x 3 panels, 3.2 m wide, tilted 35 deg toward the sun and yawed to camera
  const tilt = -0.61;
  for (let row = 0; row < 2; row++) {
    for (let col = 0; col < 3; col++) {

      const py = 1.98 + row * 1.12;

      xf(panel, { x: px, y: py, z: pz, rx: tilt, ry: arrayYaw });
      solarGeo.push(panel);

      xf(frame, { x: px, y: py, z: pz - 0.03, rx: tilt, ry: arrayYaw });

    arm.rotateX(HALF_PI);
    arm.rotateY(arrayYaw);
    arm.translate(arrayX + 0.32, 1.9 + row * 1.12 - 0.54, arrayZ);
    titanium.push(arm);
